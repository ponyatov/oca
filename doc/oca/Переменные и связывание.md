# Переменные и связывание

В OCaml концепция "переменных" существенно отличается от императивных языков. Вместо изменяемых ячеек памяти мы работаем со **связыванием имен** (name binding), что лучше соответствует функциональной парадигме.

## Иммутабельность по умолчанию

```ocaml
(* Связывание имени со значением *)
let x = 5;;                    (* x → 5 *)
let name = "OCaml";;           (* name → "OCaml" *)

(* Попытка изменить значение приведет к ошибке *)
(* x = 10;; *)                 (* Error: This is not a function *)

(* Новое связывание создает новое имя в новой области видимости *)
let x = 10 in
	x * 2;;                    (* 20, оригинальное x не изменено *)
```

## Базовый синтаксис

```ocaml
(* Глобальное связывание *)
let pi = 3.14159;;
```

## Локальные связывания с `let...in`

```ocaml
(* Глобальная область *)
let x = 10 in
	let y = 20 in
		let result = x + y in
			result (* результат выражения *)

(* После этого x, y, result больше не доступны *)
```

- Ограничение области видимости
- Локальные функции
- Безопасность: Временные переменные не "просачиваются" наружу
- Читаемость: Легче понять, где используются переменные
- Переиспользование имен: Можно использовать одинаковые имена в разных контекстах
- Управление памятью: Временные значения могут быть быстрее освобождены


```ocaml
(* Локальное связывание с ограниченной областью видимости *)
let area_of_circle radius =
  let pi = 3.14159 in
	  pi *. radius *. radius;;

(* Последовательные связывания *)
let calculate_stats numbers =
  let count = List.length numbers in
	  let sum = List.fold_left ( + ) 0 numbers in
		  let average = float_of_int sum /. float_of_int count in
			  (count, sum, average);;
```

## Императивные переменные (с использованием ссылок)

- Создание ссылки
```ocaml
let x = ref 5;;           (* x : int ref *)
let y = ref "hello";;     (* y : string ref *)
let z = ref [1; 2; 3];;   (* z : int list ref *)
```
- Получение значения
```ocaml
let value = !x;;          (* value = 5 *)
```
- Изменение значения
```ocaml
x := 10;;                 (* Теперь x содержит 10 *)
y := "world";;            (* Теперь y содержит "world" *)
```

Мутабельные переменные через ссылки позволяют хранить изменяемое состояние:
```ocaml
let counter = ref 0;;
let increment () = counter := !counter + 1;;
increment ();;                (* counter → 1 *)
increment ();;                (* counter → 2 *)
```
