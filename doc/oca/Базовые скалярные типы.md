# Базовые скалярные типы

## `int`: целые числа

```ocaml
let x : int = 42
let y = -15
let z = 0x1F  (* шестнадцатеричная запись *)
let w = 0o77  (* восьмеричная запись *)
```

Целочисленные операции:

```ocaml
let sum = 5 + 3      (* 8 *)
let diff = 10 - 4    (* 6 *)
let prod = 7 * 2     (* 14 *)
let div = 15 / 4     (* 3 - целочисленное деление *)
let mod = 15 mod 4   (* 3 *)
```

## `float`: числа с плавающей точкой

```ocaml
let pi : float = 3.14159 (* с явным указанием типа переменной *)
let e          = 2.71828 (* с выведением типа *)
let scientific = 1.23e-4 (* научная нотация *)
```

Операции с float: необходима дополнительная точка в операторах

```ocaml
let fsum  = 5.0  +. 3.0   (* 8.0 *)
let fdiff = 10.0 -. 4.0   (* 6.0 *)
let fprod = 7.0  *. 2.0   (* 14.0 *)
let fdiv  = 15.0 /. 4.0   (* 3.75 *)
```

## `bool`: логические значения
```ocaml
(* bool - логические значения *)
let is_true : bool = true
let is_false = false
```

## `char`: символы
```ocaml
(* char - символы *)
let a : char = 'a'
let newline = '\n'
let tab = '\t'
```

## `unit`: единичный тип
```ocaml
(* unit - тип с одним значением *)
let nothing : unit = ()
```

## Сравнения

```ocaml
let b1 = 5 = 5      (* true *)
let b2 = 5 <> 3     (* true *)
let b3 = 5 < 10     (* true *)
let b4 = 5.0 > 2.0  (* true *)
```

## Преобразования типов

- только явные преобразования: неявные преобразования запрещены

```ocaml
let int_to_float = float_of_int 5    (* 5.0 *)
let float_to_int = int_of_float 3.14 (* 3 *)

let char_to_int = int_of_char 'A'    (* 65 *)
let int_to_char = char_of_int 66     (* 'B' *)

let string_of_bool = string_of_bool true (* "true" *)
```
